20/// The global cooperative state.
21///
22/// Access to this state is synchronized with `WAKELOCK`. The mutex is held
23/// while changing the process mode so that no cooperating caller can observe
24/// an intermediate transition.
25struct State {
26    /// The current mode in the schedule.
27    mode: Mode,
28
29    /// How many scopes are using this schedule.
30    cooperating: usize,
31}
32
33/// The current cooperative state.
34///
35/// A single process-wide mode may have multiple cooperating scopes, including
36/// scopes belonging to different threads.
37static STATE: Mutex<State> = Mutex::new(State {
38    mode: Mode::Original,
39    cooperating: 0,
40});
41
42/// A condition variable for threads waiting for the cooperative state to
43/// become available for their requested mode.
44static WAKELOCK: Condvar = Condvar::new();
45
46thread_local! {
47    /// The modes held by `UserLock`s in this thread, from outermost to
48    /// innermost.
49    ///
50    /// This is a stack rather than a counter because a thread may own
51    /// multiple nested scopes with different modes:
52    ///
53    /// When a thread owns the entire cooperative state, changing mode pushes
54    /// the new mode onto this stack. Dropping that scope restores the mode
55    /// immediately below it.
56    static HELD: RefCell<Vec<Mode>> = const { RefCell::new(Vec::new()) };
57}
58
59pub struct UserScope;
60impl UserScope {
61    /// Acquire a new lock for the Operating Mode.
62    ///
63    /// This object is scoped such that you can guarantee that the `SetUID`
64    /// operating mode will be set to `mode` so long as this object is in scope.
65    ///
66    /// Threads that need the same mode can cooperate, and a single thread can
67    /// have multiple different modes nested within one another.
68    ///
69    /// ## Errors
70    ///
71    /// If the mode cannot be switched using the underlying
72    /// `setresuid`/`setresgid` calls.
73    ///
74    /// ## Panics
75    ///
76    /// If the underlying syscall fails
77    #[allow(clippy::significant_drop_tightening, clippy::arithmetic_side_effects)]
78    pub fn new(mode: Mode) -> Self {
79        loop {
80            let mut state = STATE.lock();
81
82            let held = HELD.with_borrow(Vec::len);
83
84            // If the state is empty or we own it, change the mode.
85            if state.cooperating == 0 || state.cooperating == held {
86                let previous = state.mode;
87                crate::switch(mode).expect("FATAL: Could not change operating mode");
88
89                HELD.with_borrow_mut(|held| held.push(previous));
90                state.mode = mode;
91                state.cooperating += 1;
92
93                mem::drop(state);
94                WAKELOCK.notify_all();
95                return Self {};
96
97            // If we can cooperate, increment the cooperative count.
98            } else if state.mode == mode {
99                // With cooperation, we push the existing state to prevent
100                // a restore in drop.
101                HELD.with_borrow_mut(|held| held.push(state.mode));
102
103                state.cooperating += 1;
104                return Self {};
105            }
106
107            // Wait for the schedule to change.
108            WAKELOCK.wait(&mut state);
109        }
110    }
111}
112
113impl Drop for UserScope {
114    #[allow(
115        clippy::significant_drop_tightening,
116        clippy::arithmetic_side_effects,
117        clippy::unwrap_used
118    )]
119    fn drop(&mut self) {
120        let mut state = STATE.lock();
121
122        let previous = HELD.with_borrow_mut(|held| held.pop().unwrap());
123
124        // Remove our scope
125        state.cooperating -= 1;
126        let empty = state.cooperating == 0;
127
128        // If the schedule is changing, revert the mode and alert others
129        if empty || state.cooperating == HELD.with_borrow(Vec::len) {
130            // Only make a syscall if we need to.
131            if state.mode != previous {
132                crate::switch(previous).expect("FAILED: Could not restore operating mode");
133                state.mode = previous;
134            }
135
136            mem::drop(state);
137
138            // If the entire state is changing, notify one thread so it can change + notify_all
139            if empty {
140                WAKELOCK.notify_one();
141
142            // If our thread has ownership, notify every other thread so they can check for
143            // cooperation.
144            } else {
145                WAKELOCK.notify_all();
146            }
147        }
148    }
149}

151/// An error when trying to change UID/GID.
152#[derive(Debug)]
153pub struct Error {
154    /// The UID we were trying to change to
155    mode: Mode,
156
157    /// The error we got from the syscall.
158    errno: Errno,
159
160    /// What syscall we tried to use
161    call: &'static str,
162}
163impl fmt::Display for Error {
164    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
165        write!(
166            f,
167            "{}: Failed to change UID to {}: {}",
168            self.call, self.mode, self.errno
169        )
170    }
171}
172impl error::Error for Error {
173    fn source(&self) -> Option<&(dyn error::Error + 'static)> {
174        Some(&self.errno as &dyn error::Error)
175    }
176}

178    #[must_use]
179    pub const fn new(mode: Mode, errno: Errno, call: &'static str) -> Self {

181    }

184/// A `SetUID` mode.
185#[derive(Debug, Copy, Clone, PartialEq, Eq)]
186pub enum Mode {
187    /// Transition to the Real user, setting both Real and Effective
188    /// to `USER.real`, while saving Effective to Saved.
189    Real,
190
191    /// Transition to the Effective user, setting both Real and Effective
192    /// to `USER.effective`, while saving Real to Saved.
193    Effective,
194
195    /// The current operating mode. This is functionally a no-op except for
196    /// in drop, where it drops whatever the current mode happens to be.
197    Existing,
198
199    /// Revert to the program's original operating mode. For set, this
200    /// mode is functionally identical to using `revert()`. For drop, it
201    /// acts as `user::revert()`.
202    Original,
203}
204impl fmt::Display for Mode {
205    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
206        match self {
207            Self::Real => write!(f, "{}", USER.real),
208            Self::Effective => write!(f, "{}", USER.effective),
209            Self::Existing => write!(f, "Existing"),
210            Self::Original => write!(f, "{}/{}/{}", USER.real, USER.effective, USER.saved),
211        }
212    }
213}

215/// Set the current operating mode.
216///
217/// Note this is *not* thread safe. If you use this while any `UserScope`s are active,
218/// you will break it.
219///
220/// ## Errors
221///
222/// If the underlying syscall fails.
223///
224/// ## Panics
225///
226/// If a `UserScope` is active in any thread
227#[allow(clippy::significant_drop_tightening)]
228pub fn set(mode: Mode) -> Result<(), Error> {
229    if !*SETUID {
230        return Ok(());
231    }
232
233    let mut lock = STATE.lock();
234    assert_eq!(
235        lock.cooperating, 0,
236        "State cannot be set while a Scope is active"
237    );
238    lock.mode = mode;
239    switch(mode)
240}

242/// Internal function to switch states.
243fn switch(mode: Mode) -> Result<(), Error> {
244    if !*SETUID {
245        return Ok(());
246    }
247    match mode {
248        Mode::Real => {
249            setresuid(USER.real, USER.real, USER.effective)
250                .map_err(|e| Error::new(mode, e, "setresuid"))?;
251            setresgid(GROUP.real, GROUP.real, GROUP.effective)
252                .map_err(|e| Error::new(mode, e, "setresgid"))?;
253        }
254        Mode::Effective => {
255            setresuid(USER.effective, USER.effective, USER.real)
256                .map_err(|e| Error::new(mode, e, "setresuid"))?;
257            setresgid(GROUP.effective, GROUP.effective, GROUP.real)
258                .map_err(|e| Error::new(mode, e, "setresgid"))?;
259        }
260        Mode::Original => {
261            setresuid(USER.real, USER.effective, USER.saved)
262                .map_err(|e| Error::new(Mode::Original, e, "setresuid"))?;
263            setresgid(GROUP.real, GROUP.effective, GROUP.saved)
264                .map_err(|e| Error::new(Mode::Original, e, "setresgid"))?;
265        }
266        Mode::Existing => {}
267    }
268
269    Ok(())
270}
271/// Destructively change mode, preventing the process from returning.
272/// This function will set Real, Effective, and Saved values to the
273/// desired Mode. This prevents the process from changing their mode
274///
275/// ## Errors
276/// If the underlying syscall fails
277pub fn drop(mode: Mode) -> Result<(), Error> {
278    match mode {
279        Mode::Real => {
280            setresuid(USER.real, USER.real, USER.real)
281                .map_err(|e| Error::new(mode, e, "setresuid"))?;
282            setresgid(GROUP.real, GROUP.real, GROUP.real)
283                .map_err(|e| Error::new(mode, e, "setresgid"))
284        }
285        Mode::Effective => {
286            setresuid(USER.effective, USER.effective, USER.effective)
287                .map_err(|e| Error::new(mode, e, "setresuid"))?;
288            setresgid(GROUP.effective, GROUP.effective, GROUP.effective)
289                .map_err(|e| Error::new(mode, e, "setresgid"))
290        }
291        Mode::Original => {
292            setresuid(USER.real, USER.effective, USER.saved)
293                .map_err(|e| Error::new(Mode::Original, e, "setresuid"))?;
294            setresgid(GROUP.real, GROUP.effective, GROUP.saved)
295                .map_err(|e| Error::new(Mode::Original, e, "setresgid"))
296        }
297        Mode::Existing => {
298            let (user, group) = (
299                getresuid().map_err(|e| Error::new(mode, e, "getresuid"))?,
300                getresgid().map_err(|e| Error::new(mode, e, "getresgid"))?,
301            );
302            setresuid(user.real, user.real, user.real)
303                .map_err(|e| Error::new(mode, e, "setresuid"))?;
304            setresgid(group.real, group.real, group.real)
305                .map_err(|e| Error::new(mode, e, "setresgid"))
306        }
307    }
308}

310/// Get the current user mode
311///
312/// Note that this is not thread-safe, and your program can suffer
313/// from TOC-TOU problems if you assume this value will remain the same
314/// when you actually need to perform a privileged operation in a multi-threaded
315/// environment.
316///
317/// ## Errors
318/// If the underlying syscall fails
319pub fn current() -> Result<Mode, Error> {
320    let uid = getresuid()
321        .map_err(|e| Error::new(Mode::Existing, e, "getresuid"))?
322        .real;
323    if uid == USER.real {
324        Ok(Mode::Real)
325    } else if uid == USER.effective {
326        Ok(Mode::Effective)
327    } else {
328        Err(Error::new(
329            Mode::Existing,
330            Errno::EINVAL,
331            "current uid unknown",
332        ))
333    }
334}

336/// This is a thread-safe wrapper that sets the mode, runs the closure/expression,
337/// then returns to the mode before the call.
338///
339/// You can use this in multi-threaded
340/// environments, and it is guaranteed the content of the closure/expression will
341/// be run under the requested Mode.
342#[macro_export]
343macro_rules! run_as {
344    ($mode:path, $ret:ty, $body:block) => {{
345        {
346            let _lock = user::UserScope::new($mode);
347            (|| -> $ret { $body })()
348        }
349    }};
350
351    ($mode:path, $body:block) => {{
352        {
353            let _lock = user::UserScope::new($mode);
354            (|| $body)()
355        }
356    }};
357
358    ($mode:path, $expr:expr) => {{
359        {
360            let _lock = user::UserScope::new($mode);
361            $expr
362        }
363    }};
364}

366#[macro_export]
367macro_rules! as_real {
368    ($ret:ty, $body:block) => {{ user::run_as!(user::Mode::Real, $ret, $body) }};
369    ($body:block) => {{ user::run_as!(user::Mode::Real, $body) }};
370    ($expr:expr) => {{ user::run_as!(user::Mode::Real, { $expr }) }};
371}

373#[macro_export]
374macro_rules! as_effective {

376    ($body:block) => {{ user::run_as!(user::Mode::Effective, $body) }};
377    ($expr:expr) => {{ user::run_as!(user::Mode::Effective, { $expr }) }};
